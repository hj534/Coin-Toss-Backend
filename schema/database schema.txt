CREATE TABLE tournament_participants (
    id SERIAL PRIMARY KEY,

    tournament_id INT NOT NULL
        REFERENCES tournaments(id)
        ON DELETE CASCADE,

    playfab_id VARCHAR(100) NOT NULL,

    display_name VARCHAR(100),

    registered_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    eliminated BOOLEAN NOT NULL DEFAULT FALSE,

    final_position INT,

    UNIQUE (tournament_id, playfab_id)
);






CREATE TABLE tournament_participants (
    id SERIAL PRIMARY KEY,

    tournament_id INT NOT NULL
        REFERENCES tournaments(id)
        ON DELETE CASCADE,

    playfab_id VARCHAR(100) NOT NULL,

    display_name VARCHAR(100),

    registered_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    eliminated BOOLEAN NOT NULL DEFAULT FALSE,

    final_position INT,

    UNIQUE (tournament_id, playfab_id)
);




CREATE TABLE tournament_matches (
    id SERIAL PRIMARY KEY,

    tournament_id INT NOT NULL
        REFERENCES tournaments(id)
        ON DELETE CASCADE,

    round_number INT NOT NULL,

    match_number INT NOT NULL,

    player1_id INT
        REFERENCES tournament_participants(id),

    player2_id INT
        REFERENCES tournament_participants(id),

    winner_id INT
        REFERENCES tournament_participants(id),

    status VARCHAR(30) NOT NULL DEFAULT 'pending',

    scheduled_start_time TIMESTAMPTZ,

    fusion_room_name VARCHAR(200),

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    completed_at TIMESTAMPTZ
);


CREATE TABLE player_leaderboard (
    playfab_id TEXT PRIMARY KEY,
    display_name TEXT NOT NULL,
    points INTEGER NOT NULL DEFAULT 0,
    active_membership_id TEXT NOT NULL DEFAULT '',
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


CREATE TABLE player_prizes (
    id SERIAL PRIMARY KEY,
    playfab_id TEXT NOT NULL,
    display_name TEXT NOT NULL,
    tournament_id INTEGER NOT NULL,
    tournament_name TEXT NOT NULL,
    rank INTEGER NOT NULL,
    prize_type TEXT NOT NULL,
    prize_name TEXT NOT NULL,
    prize_value TEXT NOT NULL DEFAULT '',
    status TEXT NOT NULL DEFAULT 'earned',
    earned_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    delivered_at TIMESTAMPTZ,
    admin_notes TEXT NOT NULL DEFAULT ''
);

CREATE UNIQUE INDEX uq_player_prizes_reward
ON player_prizes (
    playfab_id,
    tournament_id,
    rank,
    prize_type,
    prize_name
);
